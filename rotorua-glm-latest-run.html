Calibrated alum-on baseline, 1977–2025 (2026 PC10 loads): plots from the parquet output of the last local run

Author

Tadhg Moore

Published

12 November 2026

Download the GLM-AED set-up (zip, 22 MB): namelists, AED files, all boundary files (meteorology, 13 inflows, outflow) and the calibrated parameters. Add the GLM-AED alum executable to run it.

Note

This page does not run the model. The baseline is run locally (Rscript R/run_latest.R, about 8 minutes) from the rotorua-pc10/latest/ folder on the team OneDrive, which holds the calibrated GLM-AED set-up, the parameters, the executable and the observations. That run writes its full output as parquet into latest/output/, and this page syncs and plots it. The baseline has no scenarios, and the run-info table below says exactly what produced the output.

```{r setup}
#| include: false
#| echo: false
library(dplyr); library(tidyr); library(ggplot2); library(arrow); library(data.table)
library(knitr); library(kableExtra)

# where the "latest" folder lives locally: a mirror of rotorua-pc10/latest on OneDrive (synced here when the OneDrive token is set)
latest <- here::here("onedrive_mirror", "rotorua-pc10", "latest")
data_source <- "local mirror (OneDrive not used)"
if (nzchar(Sys.getenv("ONEDRIVE_TOKEN_ENCRYPTED"))) {
  source(here::here("R", "sync_aeme_onedrive.R"))
  data_source <- tryCatch({
    sync_onedrive_file("rotorua-pc10/latest/metadata.json", file.path(latest, "metadata.json"))   # only the small files: the page needs the output, not the set-up
    sync_onedrive_folder("rotorua-pc10/latest/output", file.path(latest, "output"))
    "OneDrive: rotorua-pc10/latest"
  }, error = function(e) paste0("local mirror (OneDrive sync failed: ", substr(conditionMessage(e), 1, 80), ")"))
}
out <- file.path(latest, "output"); stopifnot(dir.exists(file.path(out, "model")))
meta <- jsonlite::fromJSON(file.path(latest, "metadata.json")); runi <- jsonlite::fromJSON(file.path(out, "run_info.json"))

raw <- open_dataset(file.path(out, "model", "output_raw"))
getv <- function(v) raw |> filter(var_sim == v) |> select(date, depth, value) |> collect() |> mutate(date = as.Date(date), value = if (v == "LKE_lvlwtr") value + meta$bed_elevation_m else value)   # level is written as a depth
obs <- read_parquet(file.path(out, "observations.parquet")) |> mutate(date = as.Date(date))
sed <- read_parquet(file.path(out, "model", "sediment.parquet")) |> mutate(date = as.Date(date))   # (filtered to the output period below)
d0 <- min(getv("LKE_lvlwtr")$date); d1 <- max(getv("LKE_lvlwtr")$date); obs <- obs |> filter(date >= d0, date <= d1); sed <- sed |> filter(date >= d0)

col_obs <- "#1b3a4b"; col_sim <- "#d95f02"; col_band <- "#cfe3f1"
alum_start <- as.Date("2006-01-01")
shade <- annotate("rect", xmin = alum_start, xmax = d1 + 365, ymin = -Inf, ymax = Inf, fill = col_band, alpha = 0.4)
th <- theme_minimal(base_size = 13) + theme(legend.position = "top", strip.text = element_text(face = "bold", hjust = 0), panel.grid.minor = element_blank())

# surface (top 5 m mean) and bottom (within 3 m of the bed) daily series from the depth-resolved output
surf_bot <- function(v) {
  x <- getv(v) |> group_by(date) |> mutate(zmax = max(depth)) |> ungroup()
  bind_rows(x |> filter(depth <= 5) |> group_by(date) |> summarise(value = mean(value), .groups = "drop") |> mutate(where = "Surface (0-5 m)"),
            x |> filter(depth >= zmax - 3) |> group_by(date) |> summarise(value = mean(value), .groups = "drop") |> mutate(where = "Bottom (lowest 3 m)"))
}
obs_sb <- function(v) {
  x <- obs |> filter(var_aeme == v, !is.na(depth))
  bind_rows(x |> filter(depth <= 5) |> group_by(date) |> summarise(value = mean(value), .groups = "drop") |> mutate(where = "Surface (0-5 m)"),
            x |> filter(depth >= 17) |> group_by(date) |> summarise(value = mean(value), .groups = "drop") |> mutate(where = "Bottom (lowest 3 m)"))
}
sb_plot <- function(v, ylab) {
  s <- surf_bot(v); o <- obs_sb(v)
  ggplot() + shade + geom_line(data = s, aes(date, value, colour = "Simulated"), linewidth = 0.4) + geom_point(data = o, aes(date, value, colour = "Observed"), size = 0.7, alpha = 0.7) +
    facet_wrap(~ factor(where, c("Surface (0-5 m)", "Bottom (lowest 3 m)")), ncol = 1) + scale_colour_manual(values = c(Observed = col_obs, Simulated = col_sim)) + labs(x = NULL, y = ylab, colour = NULL) + th
}

# model vs observation pairs: simulated value at the observed date and the nearest simulated depth
pair <- function(v) {
  o <- obs |> filter(var_aeme == v); s <- getv(v)
  if (all(is.na(o$depth))) { s <- s |> group_by(date) |> summarise(sim = mean(value), .groups = "drop"); return(inner_join(o |> select(date, obs = value), s, by = "date")) }
  so <- as.data.table(s); setnames(so, "value", "sim"); setkey(so, date, depth); ot <- as.data.table(o |> filter(!is.na(depth)) |> select(date, depth, obs = value))
  ot[, sim := so[.(ot$date, ot$depth), sim, roll = "nearest"]]; ot[!is.na(sim)] |> as_tibble()
}
cal0 <- as.Date("2006-06-01"); cal1 <- as.Date("2018-06-30"); val1 <- as.Date("2022-06-30")
vars <- c(HYD_temp = "Water temperature (°C)", CHM_oxy = "Dissolved oxygen (g/m³)", LKE_lvlwtr = "Lake level (m)", NIT_tn = "Total nitrogen (g/m³)", PHS_tp = "Total phosphorus (g/m³)", PHY_tchla = "Chlorophyll-a (mg/m³)")
perf <- lapply(names(vars), function(v) { p <- pair(v)
  per <- list(`All years` = p, `Calibration 2006-06 to 2018-06` = p |> filter(date >= cal0, date <= cal1), `Held out 2018-07 to 2022-06` = p |> filter(date > cal1, date <= val1), `New years 2022-07 to 2025-12` = p |> filter(date > val1))
  bind_rows(lapply(names(per), function(k) { x <- per[[k]]; data.frame(Variable = vars[[v]], Period = k, n = nrow(x), RMSE = sqrt(mean((x$sim - x$obs)^2)), Bias = mean(x$sim - x$obs), r2 = if (nrow(x) > 2) cor(x$sim, x$obs)^2 else NA) })) }) |> bind_rows()
```
