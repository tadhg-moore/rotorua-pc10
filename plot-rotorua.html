Lake Rotorua V4 - GLM-AED output plots

  • Show All Code
  • Hide All Code
  • View Source

Rotorua V4 outputs (GLM-AED+ 4.0.0, glm4.nml, 1-year run 2008-07-01 to 2009-07-01), plotted with the AEME package rather than a hand-rolled netCDF4/matplotlib stack — this is the R/Quarto counterpart of the old plot_rotorua.ipynb.

AEME::get_var()/plot_output() cover every plain water-column and sediment-zone variable (they read output/output.nc and output/lake.csv for you, translate between AED and AEME variable names, and turn a depth-time array into a heatmap in one call). The one thing they don’t model is the dynamic sediment-heat profile (sed_heat_model = 2, a (time, nzones, sed_layers) array) — that section still opens the NetCDF file directly with ncdf4.

0. Helpers

profile_vars/zone_vars are read straight off the NetCDF file’s dimensions (AEME has no catalogue of “every variable in this output” — it only knows the variables it has a name mapping for), so this is the one place raw ncdf4 is used outside the sediment-heat section. get_var() happily accepts either the AED name ("OXY_oxy") or AEME’s own name ("CHM_oxy") for everything found this way.

Code 
nc <- nc_open(outfile)
var_dims <- sapply(names(nc$var), function(v) sapply(nc$var[[v]]$dim, `[[`, "name"))

profile_vars <- names(var_dims)[sapply(var_dims, function(d) "z" %in% d)]
zone_vars    <- names(var_dims)[sapply(var_dims, function(d) {
  "nzones" %in% d && !"z" %in% d && !"sed_layers" %in% d
})]

long_name <- function(var) {
  v <- nc$var[[var]]
  if (is.null(v)) return(var)
  sprintf("%s (%s)", if (nzchar(v$longname)) v$longname else var, v$units)
}

cat(sprintf("%d profile variables, %d zone variables\n", length(profile_vars), length(zone_vars)))

# A handful of AED variable names aren't recognised by AEME's fuzzy matcher even
# though they're just renamed internally (e.g. dissolved oxygen is `CHM_oxy`, not
# `OXY_oxy`) - resolve those explicitly rather than letting get_var() error.
var_alias <- c(OXY_oxy = "CHM_oxy")
resolve_var <- function(var) unname(ifelse(var %in% names(var_alias), var_alias[var], var))

#' GLM writes each layer against a moving Lagrangian grid, so `get_var()`'s depths
#' don't line up day to day - interpolate onto a common depth-below-surface grid
#' before tiling, or the heatmap comes out as sparse, misaligned slivers.
grid_profile <- function(df, nz = 150) {
  depths <- seq(0, max(df$depth, na.rm = TRUE), length.out = nz)
  df |>
    group_by(Date) |>
    group_modify(~ data.frame(depth = depths,
                               value = approx(.x$depth, .x$value, xout = depths, rule = 2)$y)) |>
    ungroup()
}

#' Depth-time heatmap for a (lon, lat, z, time) variable, via AEME::get_var()
plot_profile <- function(var, cmap = "viridis") {
  df <- get_var(aeme, model = "glm_aed", var_sim = resolve_var(var), remove_spin_up = FALSE) |>
    grid_profile()
  ggplot(df, aes(x = Date, y = depth, fill = value)) +
    geom_raster() +
    scale_y_reverse() +
    scale_fill_viridis_c(option = cmap, na.value = NA) +
    labs(x = NULL, y = "depth (m)", fill = NULL, title = var,
         subtitle = long_name(var)) +
    theme(plot.title = element_text(size = 10), plot.subtitle = element_text(size = 8))
}

#' One line per sediment zone for a (lon, lat, nzones, time) variable
plot_zone <- function(var) {
  df <- get_var(aeme, model = "glm_aed", var_sim = resolve_var(var), remove_spin_up = FALSE)
  ggplot(df, aes(x = Date, y = value, colour = factor(nzones))) +
    geom_line(linewidth = 0.4) +
    labs(x = NULL, y = NULL, colour = "zone", title = var, subtitle = long_name(var)) +
    theme(plot.title = element_text(size = 9), plot.subtitle = element_text(size = 7),
          legend.position = "bottom")
}

1. Lake-level summary

plot_wlev() and plot_wbal() are AEME’s built-ins for the two standard GLM diagnostics; the temperature and heat-balance panels are assembled from get_var() since AEME has no single “lake summary” plot for them.

Code 
temp_panel <- bind_rows(
  get_var(aeme, model = "glm_aed", var_sim = "surface_temp") |> mutate(series = "Surface Temp"),
  get_var(aeme, model = "glm_aed", var_sim = "max_temp")     |> mutate(series = "Max Temp"),
  get_var(aeme, model = "glm_aed", var_sim = "min_temp")     |> mutate(series = "Min Temp")
) |>
  ggplot(aes(x = Date, y = value, colour = series)) +
  geom_line(linewidth = 0.5) +
  scale_colour_manual(values = c("Surface Temp" = "firebrick", "Max Temp" = "darkorange",
                                  "Min Temp" = "steelblue")) +
  labs(title = "Surface / max / min temperature (degC)", x = NULL, y = NULL, colour = NULL) +
  theme(legend.position = "bottom")

heat_panel <- bind_rows(
  get_var(aeme, model = "glm_aed", var_sim = "LKE_Qsw") |> mutate(series = "Qsw"),
  get_var(aeme, model = "glm_aed", var_sim = "LKE_Qe")  |> mutate(series = "Qe"),
  get_var(aeme, model = "glm_aed", var_sim = "LKE_Qh")  |> mutate(series = "Qh"),
  get_var(aeme, model = "glm_aed", var_sim = "LKE_Qlw") |> mutate(series = "Qlw")
) |>
  ggplot(aes(x = Date, y = value, colour = series)) +
  geom_line(linewidth = 0.4) +
  labs(title = "Heat balance components", x = NULL, y = NULL, colour = NULL) +
  theme(legend.position = "bottom")

lake_summary <- (plot_wlev(aeme, model = "glm_aed") | plot_wbal(aeme, model = "glm_aed")) /
  (temp_panel | heat_panel)
lake_summary
ggsave(file.path(plot_dir, "lake_summary.png"), lake_summary, width = 12, height = 8, dpi = 120)

4. Dynamic sediment heat model

sed_heat_model = 2 gives every sediment zone a vertical soil-temperature profile, integrated by zZSoilTemp with the zone-average water temperature as its surface boundary condition. This is a (time, nzones, sed_layers) array with no AEME variable mapping, so it’s read directly from the NetCDF file.

What to look for: the seasonal signal should damp and lag with depth, and the deepest node should be flat.

Code 
sed_temp  <- ncvar_get(nc, "sed_temp")        # (sed_layers, nzones, time)
sed_depth <- ncvar_get(nc, "sed_layer_depth") # (sed_layers)
zone_ztemp <- ncvar_get(nc, "zone_ztemp")     # (nzones, time)
sed_hf     <- ncvar_get(nc, "sed_heatflux")   # (nzones, time)
time_vals  <- as.Date(ncvar_get(nc, "time") / 24, origin = strsplit(ncatt_get(nc, "time", "units")$value, "since ")[[1]][2])
n_zone     <- dim(sed_temp)[2]

cat("sediment nodes (m below SWI):", round(sed_depth, 3), "\n")

sed_df <- expand.grid(layer = seq_len(dim(sed_temp)[1]), zone = seq_len(n_zone), t = seq_along(time_vals)) |>
  mutate(Date = time_vals[t], depth = sed_depth[layer], value = sed_temp[cbind(layer, zone, t)])
water_df <- expand.grid(zone = seq_len(n_zone), t = seq_along(time_vals)) |>
  mutate(Date = time_vals[t], value = zone_ztemp[cbind(zone, t)])

sed_temp_plot <- ggplot(sed_df, aes(x = Date, y = value, colour = sprintf("%.2f m", depth))) +
  geom_line(linewidth = 0.4) +
  geom_line(data = water_df, aes(colour = "water"), linewidth = 0.9, colour = "black") +
  facet_wrap(~ sprintf("zone %d", zone - 1), ncol = 1, scales = "free_y") +
  labs(x = NULL, y = "degC", colour = NULL, title = "Sediment temperature profile by zone") +
  theme(legend.position = "bottom")
sed_temp_plot
ggsave(file.path(plot_dir, "sediment_temperature.png"), sed_temp_plot, width = 11, height = 3.2 * n_zone, dpi = 120)
Code 
hf_df <- expand.grid(zone = seq_len(n_zone), t = seq_along(time_vals)) |>
  mutate(Date = time_vals[t], hf = sed_hf[cbind(zone, t)],
         diff = sed_temp[cbind(1, zone, t)] - zone_ztemp[cbind(zone, t)])

hf_plot <- ggplot(hf_df, aes(x = Date, y = hf, colour = factor(zone - 1))) +
  geom_hline(yintercept = 0, colour = "grey60", linewidth = 0.3) +
  geom_line(linewidth = 0.4) +
  labs(title = "Sediment-water heat flux (W/m2; negative = water losing heat to the bed)",
       x = NULL, y = NULL, colour = "zone")

diff_plot <- ggplot(hf_df, aes(x = Date, y = diff, colour = factor(zone - 1))) +
  geom_line(linewidth = 0.4) +
  geom_hline(yintercept = 0, colour = "grey60", linewidth = 0.3) +
  labs(title = "Sediment-water temperature difference (degC)", x = NULL, y = NULL, colour = "zone")

sed_heatflux <- hf_plot / diff_plot
sed_heatflux
ggsave(file.path(plot_dir, "sediment_heatflux.png"), sed_heatflux, width = 11, height = 6, dpi = 120)

6. Save every profile variable to PNG

One figure per variable under plots/profile/. Existing files are skipped, so re-running is cheap; delete the directory to force a rebuild.

Code 
profile_dir <- file.path(plot_dir, "profile")
failed <- list()
for (var in profile_vars) {
  out_path <- file.path(profile_dir, paste0(var, ".png"))
  if (file.exists(out_path)) next
  p <- tryCatch(plot_profile(var), error = function(e) e)
  if (inherits(p, "error")) {
    failed[[var]] <- conditionMessage(p)
  } else {
    ggsave(out_path, p, width = 10, height = 4, dpi = 110)
  }
}
cat(sprintf("wrote %d profile PNGs to %s\n", length(profile_vars) - length(failed), profile_dir))
if (length(failed) > 0) print(failed)

`plot_rotorua.ipynb`.

variable (they read `output/output.nc` and `output/lake.csv` for you, translate between

## 4. Dynamic sediment heat model

`sed_heat_model = 2` gives every sediment zone a vertical soil-temperature profile,
integrated by `zZSoilTemp` with the zone-average water temperature as its surface boundary
condition. This is a `(time, nzones, sed_layers)` array with no AEME variable mapping, so
it's read directly from the NetCDF file.

What to look for: the seasonal signal should **damp and lag** with depth, and the deepest
node should be flat.

```{r}
#| label: sediment-temperature
#| fig-width: 11
#| fig-height: 10

sed_temp  <- ncvar_get(nc, "sed_temp")        # (sed_layers, nzones, time)
sed_depth <- ncvar_get(nc, "sed_layer_depth") # (sed_layers)
zone_ztemp <- ncvar_get(nc, "zone_ztemp")     # (nzones, time)
sed_hf     <- ncvar_get(nc, "sed_heatflux")   # (nzones, time)
time_vals  <- as.Date(ncvar_get(nc, "time") / 24, origin = strsplit(ncatt_get(nc, "time", "units")$value, "since ")[[1]][2])
n_zone     <- dim(sed_temp)[2]

cat("sediment nodes (m below SWI):", round(sed_depth, 3), "\n")

sed_df <- expand.grid(layer = seq_len(dim(sed_temp)[1]), zone = seq_len(n_zone), t = seq_along(time_vals)) |>
  mutate(Date = time_vals[t], depth = sed_depth[layer], value = sed_temp[cbind(layer, zone, t)])
water_df <- expand.grid(zone = seq_len(n_zone), t = seq_along(time_vals)) |>
  mutate(Date = time_vals[t], value = zone_ztemp[cbind(zone, t)])

sed_temp_plot <- ggplot(sed_df, aes(x = Date, y = value, colour = sprintf("%.2f m", depth))) +
  geom_line(linewidth = 0.4) +
  geom_line(data = water_df, aes(colour = "water"), linewidth = 0.9, colour = "black") +
  facet_wrap(~ sprintf("zone %d", zone - 1), ncol = 1, scales = "free_y") +
  labs(x = NULL, y = "degC", colour = NULL, title = "Sediment temperature profile by zone") +
  theme(legend.position = "bottom")
sed_temp_plot
ggsave(file.path(plot_dir, "sediment_temperature.png"), sed_temp_plot, width = 11, height = 3.2 * n_zone, dpi = 120)
```

#| label: sediment-heatflux

hf_df <- expand.grid(zone = seq_len(n_zone), t = seq_along(time_vals)) |>

## 6. Save every profile variable to PNG

One figure per variable under `plots/profile/`. Existing files are skipped, so re-running
is cheap; delete the directory to force a rebuild.

```{r}
#| label: dump-profile
#| message: false
#| warning: false

profile_dir <- file.path(plot_dir, "profile")
failed <- list()
for (var in profile_vars) {
  out_path <- file.path(profile_dir, paste0(var, ".png"))
  if (file.exists(out_path)) next
  p <- tryCatch(plot_profile(var), error = function(e) e)
  if (inherits(p, "error")) {
    failed[[var]] <- conditionMessage(p)
  } else {
    ggsave(out_path, p, width = 10, height = 4, dpi = 110)
  }
}
cat(sprintf("wrote %d profile PNGs to %s\n", length(profile_vars) - length(failed), profile_dir))
if (length(failed) > 0) print(failed)
```
